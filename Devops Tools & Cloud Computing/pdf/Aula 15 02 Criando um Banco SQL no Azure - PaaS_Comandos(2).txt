-- Slide 20

az sql db ltr-policy set \
  --resource-group rg-sql-dimdim \
  --server sql-server-dimdim-rm9999-eastus2 \
  --name db-dimdim \
  --weekly-retention P30D \
  --monthly-retention P365D \
  --yearly-retention P1825D \
  --week-of-year 1

Descrição:
--weekly-retention P30D: Mantém os backups semanais armazenados por 1 mês
--monthly-retention P365D: Mantém os backups mensais armazenados por 12 meses (1 ano)
--yearly-retention P1825D: Mantém os backups anuais armazenados por 5 anos
--week-of-year 1 : Indica que o backup anual será tirado na primeira semana do ano

-- Slide 37

az sql failover-group create \
  --name failover-group-dimdim-rm9999 \
  --resource-group rg-sql-dimdim \
  --server sql-server-dimdim-rm9999-eastus2 \
  --partner-server sql-server-dimdim-rm9999-westus2 \
  --partner-resource-group rg-sql-dimdim \
  --failover-policy Automatic \
  --grace-period 1


Descrição:
--grace-period é o tempo (em minutos e no mínimo 60) que o Azure espera após detectar uma falha no Servidor Primário antes de iniciar automaticamente o Failover para o Servidor Secundário (Parceiro)


-- Listar os Servidores do Grupo de Failover
az sql failover-group show --name failover-group-dimdim-rm9999 --resource-group rg-sql-dimdim --server sql-server-dimdim-rm9999-eastus2
