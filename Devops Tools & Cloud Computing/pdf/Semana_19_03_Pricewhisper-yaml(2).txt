trigger:
  branches:
    include:
      - Main

variables:
  buildConfiguration: 'Release'
  azureSubscription: 'MyAzureSubscription'
  apiProject: 'pricewhisper-usuarios.API/pricewhisper-usuarios.API.csproj'
  artifactName: 'app'

stages:

# =========================================================
# 1. CRIAR INFRAESTRUTURA
# =========================================================
- stage: Infraestrutura
  displayName: 'Criar infraestrutura'
  jobs:
    - job: CriarInfra
      displayName: 'Provisionar recursos Azure'
      pool:
        vmImage: 'ubuntu-latest'

      steps:
        - checkout: self

        - task: AzureCLI@2
          displayName: 'Executar script de infraestrutura'
          inputs:
            azureSubscription: '$(azureSubscription)'
            scriptType: 'bash'
            scriptLocation: 'scriptPath'
            scriptPath: 'pricewhisper-usuarios.API/infra-app.sh'
            arguments: '"$(ORACLE_HOST)" "$(ORACLE_PORT)" "$(ORACLE_SID)" "$(ORACLE_USER)" "$(ORACLE_PASS)"'
            workingDirectory: '$(Build.SourcesDirectory)'
            failOnStandardError: false

# =========================================================
# 2. BUILD, TESTES E PUBLICAÇÃO DO PACOTE
# =========================================================
- stage: BuildTestes
  displayName: 'Build e testes'
  dependsOn: Infraestrutura
  condition: succeeded()

  jobs:
    - job: BuildTest
      displayName: 'Compilar e testar API'
      pool:
        vmImage: 'windows-latest'

      steps:
        - checkout: self

        - task: DotNetCoreCLI@2
          displayName: 'Restaurar dependências'
          inputs:
            command: 'restore'
            projects: '$(apiProject)'

        - task: DotNetCoreCLI@2
          displayName: 'Build da API'
          inputs:
            command: 'build'
            projects: '$(apiProject)'
            arguments: '--configuration $(buildConfiguration)'

        - task: DotNetCoreCLI@2
          displayName: 'Testes de CRUD'
          inputs:
            command: 'test'
            projects: '**/*Tests.csproj'
            arguments: '--configuration $(buildConfiguration)'
            testRunTitle: 'Testes de CRUD'

        - task: DotNetCoreCLI@2
          displayName: 'Publicar API'
          inputs:
            command: 'publish'
            publishWebProjects: false
            projects: '$(apiProject)'
            arguments: '--configuration $(buildConfiguration) --output $(Build.ArtifactStagingDirectory)/publish'
            zipAfterPublish: false
            modifyOutputPath: false

        - task: ArchiveFiles@2
          displayName: 'Compactar aplicação'
          inputs:
            rootFolderOrFile: '$(Build.ArtifactStagingDirectory)/publish'
            includeRootFolder: false
            archiveType: 'zip'
            archiveFile: '$(Build.ArtifactStagingDirectory)/$(artifactName).zip'
            replaceExistingArchive: true

        - task: PublishPipelineArtifact@1
          displayName: 'Publicar artefato'
          inputs:
            targetPath: '$(Build.ArtifactStagingDirectory)/$(artifactName).zip'
            artifact: '$(artifactName)'
            publishLocation: 'pipeline'

# =========================================================
# 3. DEPLOY NO AZURE APP SERVICE
# =========================================================
- stage: Deploy
  displayName: 'Deploy no Azure App Service'
  dependsOn: BuildTestes
  condition: succeeded()

  jobs:
    - job: DeployWebApp
      displayName: 'Implantar API'
      pool:
        vmImage: 'windows-latest'

      steps:
        - checkout: none

        - task: DownloadPipelineArtifact@2
          displayName: 'Baixar artefato'
          inputs:
            buildType: 'current'
            artifactName: '$(artifactName)'
            targetPath: '$(Pipeline.Workspace)/$(artifactName)'

        - task: AzureAppServiceSettings@1
          displayName: 'Configurar variáveis do App Service'
          inputs:
            azureSubscription: '$(azureSubscription)'
            appName: '$(NOME_WEBAPP)'
            appSettings: |
              [
                {
                  "name": "ASPNETCORE_ENVIRONMENT",
                  "value": "$(ASPNETCORE_ENVIRONMENT)",
                  "slotSetting": false
                },
                {
                  "name": "ConnectionStrings__OracleConnection",
                  "value": "$(OracleConnection)",
                  "slotSetting": false
                }
              ]

        - task: AzureWebApp@1
          displayName: 'Deploy da API'
          inputs:
            azureSubscription: '$(azureSubscription)'
            appType: 'webApp'
            appName: '$(NOME_WEBAPP)'
            package: '$(Pipeline.Workspace)/$(artifactName)/$(artifactName).zip'
            deploymentMethod: 'zipDeploy'
